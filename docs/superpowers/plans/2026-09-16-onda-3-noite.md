# config/test_cartoes.py
"""Cartão dentro de cartão é moldura repetida — o defeito que a T3.3 do plano
mestre existe para impedir ("três caixas com borda dentro de uma caixa com
borda"). Antes de trocar a pele de 114 cartões, medir: hoje, quantos nascem
aninhados? A resposta vira catraca."""
import importlib
from html.parser import HTMLParser

from django.test import TestCase

from plans.tests import create_complete_user


class _Aninhamento(HTMLParser):
    VAZIOS = {"br", "img", "input", "meta", "link", "hr", "source", "use", "path", "circle", "rect", "line"}

    def __init__(self):
        super().__init__()
        self.pilha = []  # True quando o elemento aberto é um .card
        self.aninhados = []

    def handle_starttag(self, tag, attrs):
        if tag in self.VAZIOS:
            return
        classes = dict(attrs).get("class", "").split()
        eh_card = "card" in classes
        if eh_card and any(self.pilha):
            self.aninhados.append(" ".join(classes))
        self.pilha.append(eh_card)

    def handle_endtag(self, tag):
        if tag in self.VAZIOS or not self.pilha:
            return
        self.pilha.pop()


def cartoes_aninhados(html):
    p = _Aninhamento()
    p.feed(html)
    return p.aninhados


class CartaoDentroDeCartaoTests(TestCase):
    ROTAS = ("/", "/treino/", "/historico/", "/hidratacao/", "/areas/", "/conquistas/", "/conta/perfil/")

    #: Catraca: o que se mede hoje. Só desce.
    TETO_ANINHADOS = 0

    def setUp(self):
        self.user = create_complete_user("cartoes@exemplo.com")
        self.client.force_login(self.user)

    def test_o_leitor_enxerga_um_aninhamento(self):
        """Controle positivo do leitor: sem isto, um HTML sem `.card` deixaria o
        teste verde por não achar nada."""
        self.assertEqual(cartoes_aninhados('<div class="card a"><section class="card b"></section></div>'), ["card b"])
        self.assertEqual(cartoes_aninhados('<div class="card a"></div><div class="card b"></div>'), [])

    def test_nenhuma_rota_renderiza_cartao_dentro_de_cartao(self):
        achados = {}
        for rota in self.ROTAS:
            resposta = self.client.get(rota, follow=True)
            self.assertEqual(resposta.status_code, 200, rota)
            aninhados = cartoes_aninhados(resposta.content.decode())
            if aninhados:
                achados[rota] = aninhados
        total = sum(len(v) for v in achados.values())
        self.assertLessEqual(total, self.TETO_ANINHADOS, f"cartões aninhados: {achados}")
```

no CSS REAL (não na spec), Mesa e Ferro.

    .venv/Scripts/python.exe scripts/qa/auditar_contraste.py > scratchpad/shots-design/contraste.md

Par que reprova NÃO é corrigido aqui: vira item de BACKLOG com a correção
proposta — cor é decisão de direção."""
import os
import sys
from pathlib import Path

RAIZ = Path(__file__).resolve().parent.parent.parent
sys.path.insert(0, str(RAIZ))
os.environ.setdefault("DJANGO_SETTINGS_MODULE", "config.settings")
import django  # noqa: E402

django.setup()
from config.tests import _contraste, _tokens  # noqa: E402

TEXTOS = ("--text", "--text-dim", "--text-mute", "--brand", "--agua-texto", "--brasa", "--terra", "--danger")
FUNDOS = ("--bg", "--surface", "--surface-2", "--surface-3", "--surface-focus", "--brand-soft", "--terra-soft", "--danger-soft")
GRAFICOS = ("--folha", "--agua", "--brasa", "--terra", "--chama", "--carb", "--fat", "--dia-a", "--dia-b", "--dia-c", "--dia-d", "--dia-e")
SUPERFICIES = ("--bg", "--surface", "--surface-2", "--surface-3")
ESPECIAIS = (("--on-brand", "--brand", 4.5),)
MINIMO_TEXTO, MINIMO_GRAFICO = 4.5, 3.0


def auditar(css):
    temas = {"Mesa": _tokens(css, ":root {"), "Ferro": _tokens(css, "prefers-color-scheme: dark) {" + chr(10) + "  :root {")}
    linhas, reprovados = [], []
    for tema, t in temas.items():
        pares = [(a, b, MINIMO_TEXTO) for a in TEXTOS for b in FUNDOS] + [(a, b, MINIMO_GRAFICO) for a in GRAFICOS for b in SUPERFICIES] + list(ESPECIAIS)
        for cor, fundo, minimo in pares:
            if cor not in t or fundo not in t:
                continue
            razao = _contraste(t[cor], t[fundo])
            ok = razao >= minimo
            linhas.append(f"| {tema} | `{cor}` {t[cor]} | `{fundo}` {t[fundo]} | {razao:.2f} | {minimo} | {'ok' if ok else '**REPROVA**'} |")
            if not ok:
                reprovados.append((tema, cor, fundo, razao, minimo))
    return linhas, reprovados


if __name__ == "__main__":
    css = (RAIZ / "static" / "css" / "app.css").read_text(encoding="utf-8")
    linhas, reprovados = auditar(css)
    print("| tema | texto/gráfico | fundo | razão | mínimo | veredito |")
    print("|---|---|---|---:|---:|---|")
    print("\n".join(linhas))
    print(f"\n**{len(linhas)} pares, {len(reprovados)} reprovados.**")
    for tema, cor, fundo, razao, minimo in reprovados:
        print(f"- {tema}: `{cor}` sobre `{fundo}` = {razao:.2f} (mínimo {minimo})")
```
